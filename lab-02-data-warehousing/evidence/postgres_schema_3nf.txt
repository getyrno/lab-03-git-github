    vc.inventory_code,
    f.title,

    ri.actual_return_date,
    ri.daily_rate,
    GREATEST(1, COALESCE(ri.actual_return_date, lab."current_date"()) - r.issue_date) AS billed_days,
    ri.daily_rate * GREATEST(1, COALESCE(ri.actual_return_date, lab."current_date"()) - r.issue_date)::numeric AS item_cost,

        END AS overdue_days
