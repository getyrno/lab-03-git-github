     JOIN ( SELECT DISTINCT ON (sat_video_copy_details.copy_hk) sat_video_copy_details.copy_hk,
            sat_video_copy_details.load_dts,
            sat_video_copy_details.hashdiff,
            sat_video_copy_details.condition,
            sat_video_copy_details.storage_location,
            sat_video_copy_details.record_source
           FROM video_rental_dv.sat_video_copy_details
          ORDER BY sat_video_copy_details.copy_hk, sat_video_copy_details.load_dts DESC) sd ON sd.copy_hk = hvc.copy_hk
     JOIN ( SELECT DISTINCT ON (sat_video_copy_acquisition.copy_hk) sat_video_copy_acquisition.copy_hk,
            sat_video_copy_acquisition.load_dts,
            sat_video_copy_acquisition.hashdiff,
            sat_video_copy_acquisition.acquisition_date,
            sat_video_copy_acquisition.purchase_price,
            sat_video_copy_acquisition.record_source
           FROM video_rental_dv.sat_video_copy_acquisition
          ORDER BY sat_video_copy_acquisition.copy_hk, sat_video_copy_acquisition.load_dts DESC) sa ON sa.copy_hk = hvc.copy_hk;
